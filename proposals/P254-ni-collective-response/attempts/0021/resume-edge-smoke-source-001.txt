def require(name,value):
    if not value: raise AssertionError(name)

def refused(name,fn):

def kill_publish(path,value):
    pid=os.fork()
    if pid==0:
        original=os.replace
        def stop_after_rotation(source,destination):
            original(source,destination)
            if Path(destination)==path.with_suffix(path.suffix+'.prev'):
                os.kill(os.getpid(),signal.SIGSTOP)
        os.replace=stop_after_rotation
        save_json(path,value)
        os._exit(3)
    try:
        _,status=os.waitpid(pid,os.WUNTRACED)
        if not os.WIFSTOPPED(status): raise AssertionError('child did not reach rotation boundary')
    finally:
        os.kill(pid,signal.SIGKILL)
        os.waitpid(pid,0)
