#!/usr/bin/env bash
set -u
cd /tmp/substrate-particles
attempt=proposals/P253-euler-particle-mechanisms/attempts/0060
python=/home/dan/substrate-framework/.venv/bin/python
cmd="PYTHONPATH=src $python $attempt/verify_neutral_cell.py"
printf '%s\n' "$cmd" > "$attempt/multipole-order-correction.command.txt"
: > "$attempt/multipole-order-correction.stderr"
PYTHONPATH=src "$python" "$attempt/verify_neutral_cell.py" > "$attempt/multipole-order-correction.stdout" 2> "$attempt/multipole-order-correction.stderr"
verify_rc=$?
printf '%s\n' "$verify_rc" > "$attempt/multipole-order-correction.exit"
tests="PYTHONPATH=src $python -m pytest -q tests/test_euler_neutral_cell.py"
printf '%s\n' "$tests" > "$attempt/focused-pytest-repaired.command.txt"
: > "$attempt/focused-pytest-repaired.stderr"
PYTHONPATH=src "$python" -m pytest -q tests/test_euler_neutral_cell.py > "$attempt/focused-pytest-repaired.stdout" 2> "$attempt/focused-pytest-repaired.stderr"
test_rc=$?
printf '%s\n' "$test_rc" > "$attempt/focused-pytest-repaired.exit"
printf 'P253_0060_REPAIRED verify=%s tests=%s\n' "$verify_rc" "$test_rc"
if [ "$verify_rc" -ne 0 ] || [ "$test_rc" -ne 0 ]; then
  exit 1
fi
