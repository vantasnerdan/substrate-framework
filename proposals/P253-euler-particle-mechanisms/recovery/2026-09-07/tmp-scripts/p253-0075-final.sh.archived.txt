#!/usr/bin/env bash
set -u
cd /tmp/substrate-particles
ATTEMPT=proposals/P253-euler-particle-mechanisms/attempts/0075
PY=/tmp/substrate-particles/.venv/bin/python
printf '%s\n' 'PYTHONPATH=src .venv/bin/python proposals/P253-euler-particle-mechanisms/attempts/0075/verify_weighted_tail.py' > "$ATTEMPT/weighted-tail.command.txt"
PYTHONPATH=src "$PY" "$ATTEMPT/verify_weighted_tail.py" > "$ATTEMPT/weighted-tail.stdout" 2> "$ATTEMPT/weighted-tail.stderr"
status=$?
printf '%s\n' "$status" > "$ATTEMPT/weighted-tail.exit"
if [ "$status" -ne 0 ]; then exit "$status"; fi
printf '%s\n' 'PYTHONPATH=src .venv/bin/python -m pytest -q tests/test_euler_asymptotic_tails.py' > "$ATTEMPT/focused-tests.command.txt"
PYTHONPATH=src "$PY" -m pytest -q tests/test_euler_asymptotic_tails.py > "$ATTEMPT/focused-tests.stdout" 2> "$ATTEMPT/focused-tests.stderr"
status=$?
printf '%s\n' "$status" > "$ATTEMPT/focused-tests.exit"
exit "$status"
